% CS527 Lab Program 4: Vector Addition of Two Arrays A and B of size N
% N is stored at address 0 (multiple of 8)
% Array A starts at address 4 (size 4*N bytes)
% Array B starts at address 4 + 4*N (size 4*N bytes)
% Result array starts at address 4 + 8*N (size 4*N bytes)
% Uses vector SIMD instructions for addition

x1 = 0
x2 = [x1]        % load N from address 0

% Calculate base addresses:
x3 = 4           % base address of Array A
x4 = x2 * 4      % 4*N
x5 = x3 + x4     % base address of Array B = 4 + 4*N
x6 = x5 + x4     % base address of Result Array C = 4 + 8*N

x7 = 0           % loop index i = 0

.vec_add_loop
x10 = x7 - x2    % compare index i with N
BEQ .done

v1 = [x3]        % vector load 8 words from A (x3 auto-increments by 32)
v2 = [x5]        % vector load 8 words from B (x5 auto-increments by 32)
v3 = v1 + v2     % SIMD parallel 8-way addition
[x6] = v3        % vector store 8 words to C (x6 auto-increments by 32)

x7 = x7 + 8      % advance index by 8
BAL .vec_add_loop

.done
print x7         % print number of processed elements
print x6         % print final result pointer
