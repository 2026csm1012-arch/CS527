% Basic Arithmetic, Memory & Control Test Program
% Tests integer math, registers, constants, and flags
% Data Movement
x1 = 25
x2 = 10
% Arithmetic
x3 = x1 + x2   % x3 = 35
print x3

x4 = x1 - x2   % x4 = 15
print x4

x5 = x1 * 4    % x5 = 100
print x5

x6 = x5 / 5    % x6 = 20
print x6

% Memory store and load via brackets
x7 = 64
[x7] = x3      % write 35 to address 64
x8 = [x7]      % read back into x8
print x8

% Conditional branching test
x9 = x8 - 35   % sets Z flag (result is 0)
BEQ .branch_success
x10 = 0        % should not execute
print x10

.branch_success
x10 = 200
print x10      % should output 200 (0xC8)
